cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "CUDA architecture")
endif()
project(libnvm LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 11)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)

file(GLOB nvm_sources CONFIGURE_DEPENDS src/*.cpp src/linux/*.cpp)
add_library(nvm SHARED ${nvm_sources})
target_include_directories(nvm PUBLIC include include/freestanding/include PRIVATE src)
target_compile_definitions(nvm PRIVATE _CUDA __CUDA__)
target_link_libraries(nvm PUBLIC CUDA::cudart CUDA::cuda_driver Threads::Threads)

add_executable(gpu-io-benchmark benchmarks/io/gpu_io_benchmark.cu)
target_compile_definitions(gpu-io-benchmark PRIVATE __CUDA__ IO_SCHED_EVICT IO_SCHED_GLOBAL_SORT=2
  IO_SCHED_KNEE_LIMIT=32 AE_CORE_MAX_WINDOW=256)
target_compile_options(gpu-io-benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_link_libraries(gpu-io-benchmark PRIVATE nvm)

option(AE_BUILD_GDS "Build the cuFile baseline" ON)
if(AE_BUILD_GDS)
  find_path(CUFILE_INCLUDE cufile.h HINTS ${CUDAToolkit_INCLUDE_DIRS} REQUIRED)
  find_library(CUFILE_LIBRARY cufile HINTS "${CUDAToolkit_LIBRARY_DIR}"
    "${CUDAToolkit_LIBRARY_ROOT}/lib64" REQUIRED)
  add_executable(gds-io-benchmark benchmarks/io/gds_io_benchmark.cu)
  target_include_directories(gds-io-benchmark PRIVATE "${CUFILE_INCLUDE}")
  target_link_libraries(gds-io-benchmark PRIVATE CUDA::cudart "${CUFILE_LIBRARY}" Threads::Threads)
endif()

option(AE_BUILD_KERNEL_MODULE "Build the libnvm kernel module" OFF)
set(NVIDIA "" CACHE PATH "Directory containing nv-p2p.h")
set(NVIDIA_SYMVERS "" CACHE FILEPATH "NVIDIA Module.symvers")
set(KERNEL "/lib/modules/${CMAKE_SYSTEM_VERSION}/build" CACHE PATH "Kernel build directory")
if(AE_BUILD_KERNEL_MODULE)
  if(NOT EXISTS "${NVIDIA}/nv-p2p.h" OR NOT EXISTS "${NVIDIA_SYMVERS}")
    message(FATAL_ERROR "Set NVIDIA and NVIDIA_SYMVERS for the installed NVIDIA driver")
  endif()
  find_program(KBUILD_MAKE make REQUIRED)
  set(module_root "${CMAKE_CURRENT_SOURCE_DIR}/module")
  set(module_output "${CMAKE_BINARY_DIR}/module")
  set(module_ccflags "-D_CUDA -I${NVIDIA} -I${CMAKE_CURRENT_SOURCE_DIR}/src -I${CMAKE_CURRENT_SOURCE_DIR}/src/linux")
  set(module_symbols "${NVIDIA_SYMVERS}")
  configure_file(module/Makefile.in "${module_output}/Makefile" @ONLY)
  add_custom_target(kernel_module COMMAND "${KBUILD_MAKE}" WORKING_DIRECTORY "${module_output}")
endif()

enable_testing()
add_executable(test-io-workload tests/test_io_workload.cpp)
add_test(NAME io-workload COMMAND test-io-workload)
